# Bugfix: v1.13.0 — pass VAULT_TOKEN via stdin, never process argv

**Branch:** `k3d-manager-v1.13.0`
**Files:** `scripts/plugins/vault.sh`, `scripts/tests/plugins/vault_app_auth_enable_idempotent.bats`, `scripts/tests/plugins/vault_token_stdin.bats` (new)

---

## Problem

Every privileged Vault command run inside the pod carries the session (or root)
token as an `env VAULT_TOKEN=<token>` assignment on the pod exec argv. The token
is kept out of the *logged* `sh -lc` string, but it is still visible in the
**pod's process argv** — anyone with `ps` / `/proc/*/cmdline` access on the node
(or a co-scheduled container) can read a live Vault token.

Three sites expose it:

| Site | Line | Path |
|------|------|------|
| `_vault_exec` | 209 | `-- env VAULT_TOKEN=$session_token sh -lc "$cmd"` |
| `_vault_exec_stream` (session branch) | 164 / 170 / 177 | prepends `env VAULT_TOKEN=$session_token` to the exec argv |
| `_vault_login` (token verify) | 1600 | inlines `env VAULT_TOKEN="$token" vault token lookup` before the session token is cached |

**Root cause:** the token is placed in the command's argument vector, which is
world-readable on the node; it must instead be delivered on stdin (which
`kubectl exec -i` already forwards) and read into the environment inside the pod
before the real command runs.

---

## Reproduction

```bash
# With a cached session token, run any vault command through _vault_exec, and
# inspect the kubectl argv that would be sent to the API server.
# Expected (after fix): no "VAULT_TOKEN=" anywhere in the kubectl arguments.
# Actual  (before fix): the exec argv still carried  env VAULT_TOKEN=hvs.xxxx sh -lc "vault ..."
```

Covered deterministically by the BATS tests below (no live cluster).

---

## Design

Deliver the token as the **first line of stdin**. A tiny pod-side shell reads
that one line into `VAULT_TOKEN`, exports it, then `exec`s the real command,
which continues reading any remaining stdin (e.g. a policy piped to
`vault policy write -`).

- `printf` is a bash **builtin**, so writing the token to a pipe does **not**
  expose it in the local process argv either.
- On Alpine/BusyBox `sh` (the HashiCorp Vault image), `read` consumes exactly up
  to the first newline from a pipe and leaves the remainder on fd 0 for the
  `exec`ed command — this is what lets a piped policy still reach
  `vault policy write -` after the token line is stripped.
- `__vault_exec_kubectl` retries on "container not found". Since stdin is
  consumed on the first attempt, it must **buffer** stdin once and **replay** it
  on every retry — otherwise a retry sends an empty token. A new `--exec-stdin`
  flag opts a call into this buffering; callers that pass no stdin are
  unaffected (no blocking read).
- `_vault_exec_stream` gains a `--stdin` flag so callers that already pipe a
  payload (heredoc policies, `printf | ...`) opt into multiplexing: the token
  line is emitted first, then their payload is forwarded via `cat`.

## Fix

### Change 1 — `__vault_exec_kubectl`: buffer + replay stdin on `--exec-stdin`

**Exact old block (lines 36–47):**

```bash
function __vault_exec_kubectl() {
   local use_container="${1:-0}" ns="${2:-$VAULT_NS_DEFAULT}" release="${3:-$VAULT_RELEASE_DEFAULT}"
   shift 3
   local -a cmd_args=("$@")
   local output rc
   local retries=0
   local max_retries=5
   local container_removed=0

   while :; do
      output=$(_kubectl "${cmd_args[@]}" 2>&1)
      rc=$?
```

**Exact new block:**

```bash
function __vault_exec_kubectl() {
   local buffer_stdin=0
   if [[ "${1:-}" == "--exec-stdin" ]]; then
      buffer_stdin=1
      shift
   fi
   local use_container="${1:-0}" ns="${2:-$VAULT_NS_DEFAULT}" release="${3:-$VAULT_RELEASE_DEFAULT}"
   shift 3
   local -a cmd_args=("$@")
   local output rc
   local retries=0
   local max_retries=5
   local container_removed=0
   local stdin_payload=""
   if (( buffer_stdin )); then
      stdin_payload=$(cat; printf 'x')
      stdin_payload=${stdin_payload%x}
   fi

   while :; do
      if (( buffer_stdin )); then
         output=$(printf '%s' "$stdin_payload" | _kubectl "${cmd_args[@]}" 2>&1)
      else
         output=$(_kubectl "${cmd_args[@]}" 2>&1)
      fi
      rc=$?
```

> The `cat; printf 'x'` / `%x` trick preserves any trailing newlines in the
> buffered payload (command substitution would otherwise strip them).

### Change 3 — `_vault_exec_stream`: token-via-stdin session branch

**Exact old block (lines 159–183):**

```bash
   local session_token="${_VAULT_SESSION_TOKENS[$key]:-}"
   if [[ -n "$session_token" ]]; then
      local -a cmd_args=()
      local inserted=0
      if (( $# == 0 )); then
         cmd_args+=("env" "VAULT_TOKEN=$session_token")
      else
         while [[ $# -gt 0 ]]; do
            local arg="$1"
            shift
            if (( !inserted )) && [[ "$arg" == "--" ]]; then
               cmd_args+=("$arg" "env" "VAULT_TOKEN=$session_token")
               inserted=1
               continue
            fi
            cmd_args+=("$arg")
         done
         if (( !inserted )); then
            cmd_args=("env" "VAULT_TOKEN=$session_token" "${cmd_args[@]}")
         fi
      fi
      __vault_exec_kubectl "$use_container" "$ns" "$release" "${args[@]}" "${cmd_args[@]}"
   else
      __vault_exec_kubectl "$use_container" "$ns" "$release" "${args[@]}" "$@"
   fi
```

**Exact new block:**

```bash
   local session_token="${_VAULT_SESSION_TOKENS[$key]:-}"
   if [[ -n "$session_token" ]]; then
      local -a pre_dd=() remote_cmd=()
      local seen_dd=0
      while [[ $# -gt 0 ]]; do
         local arg="$1"
         shift
         if (( !seen_dd )) && [[ "$arg" == "--" ]]; then
            seen_dd=1
            continue
         fi
         if (( seen_dd )); then
            remote_cmd+=("$arg")
         else
            pre_dd+=("$arg")
         fi
      done
      if (( !seen_dd )); then
         remote_cmd=("${pre_dd[@]}")
         pre_dd=()
      fi
      local reader='IFS= read -r __VAULT_TOKEN; export VAULT_TOKEN="$__VAULT_TOKEN"; unset __VAULT_TOKEN; exec "$@"'
      if (( want_stdin )); then
         { printf '%s\n' "$session_token"; cat; } | \
            __vault_exec_kubectl --exec-stdin "$use_container" "$ns" "$release" \
               "${args[@]}" "${pre_dd[@]}" -- sh -c "$reader" _ "${remote_cmd[@]}"
      else
         printf '%s\n' "$session_token" | \
            __vault_exec_kubectl --exec-stdin "$use_container" "$ns" "$release" \
               "${args[@]}" "${pre_dd[@]}" -- sh -c "$reader" _ "${remote_cmd[@]}"
      fi
   else
      if (( want_stdin )); then
         __vault_exec_kubectl --exec-stdin "$use_container" "$ns" "$release" "${args[@]}" "$@"
      else
         __vault_exec_kubectl "$use_container" "$ns" "$release" "${args[@]}" "$@"
      fi
   fi
```

### Change 5 — `_vault_login`: cache token first, verify via stdin

**Exact old block (lines 1598–1607):**

```bash
  local login_rc=0
  local login_output=""
  login_output=$(_no_trace _vault_exec_stream --no-exit "$ns" "$release" -- env VAULT_TOKEN="$token" vault token lookup -format=json 2>&1) || login_rc=$?
  if (( login_rc != 0 )); then
      login_output=${login_output//$'\n'/ }
      login_output=${login_output//$'\r'/ }
      _err "[vault] failed to login to ${release} in namespace ${ns}: ${login_output:-unknown error}"
  fi

  _VAULT_SESSION_TOKENS["${ns}/${release}"]="$token"
```

**Exact new block:**

```bash
  local login_rc=0
  local login_output=""
  _VAULT_SESSION_TOKENS["${ns}/${release}"]="$token"
  login_output=$(_no_trace _vault_exec_stream --no-exit "$ns" "$release" -- vault token lookup -format=json 2>&1) || login_rc=$?
  if (( login_rc != 0 )); then
      unset _VAULT_SESSION_TOKENS["${ns}/${release}"]
      login_output=${login_output//$'\n'/ }
      login_output=${login_output//$'\r'/ }
      _err "[vault] failed to login to ${release} in namespace ${ns}: ${login_output:-unknown error}"
  fi
```

> Caching the token *before* the lookup lets the standard session branch deliver
> it via stdin; on failure it is unset before `_err`.

### Change 6 — add `--stdin` to every `_vault_exec_stream` caller that pipes a payload

These callers pipe stdin (heredoc or `printf | ...`). Add `--stdin` immediately
after `--no-exit` so the token line is multiplexed ahead of the payload. The
**non-piping** callers (lines 487, 1511, 1853, 1883, 1915) are left unchanged.

### Change 8 — new BATS coverage: `scripts/tests/plugins/vault_token_stdin.bats`

Create a focused suite proving the token never lands in argv for all three
paths and that `--exec-stdin` replays stdin across a retry.

```bash
#!/usr/bin/env bats

setup() {
  source "${BATS_TEST_DIRNAME}/../test_helpers.bash"
  SCRIPT_DIR="${BATS_TEST_DIRNAME}/../.."
  PLUGINS_DIR="$BATS_TEST_TMPDIR/plugins"
  mkdir -p "$PLUGINS_DIR"

  _err()  { printf 'ERROR: %s\n' "$*" >&2; exit 1; }
  _warn() { printf 'WARN: %s\n' "$*" >&2; }
  _info() { printf 'INFO: %s\n' "$*" >&2; }
  _no_trace() { "$@"; }
  export -f _err _warn _info _no_trace

  touch "$PLUGINS_DIR/eso.sh"
  mkdir -p "$SCRIPT_DIR/lib"
  touch "$SCRIPT_DIR/lib/vault_pki.sh"

  # shellcheck disable=SC1090
  source "${SCRIPT_DIR}/plugins/vault.sh"

  _vault_container_name() { printf '%s\n' "vault"; }
  export VAULT_NS_DEFAULT="secrets"
  export VAULT_RELEASE_DEFAULT="vault"

  ARGV_LOG="$BATS_TEST_TMPDIR/argv.log"
  STDIN_LOG="$BATS_TEST_TMPDIR/stdin.log"
  : >"$ARGV_LOG"
  : >"$STDIN_LOG"
}

@test "_vault_exec keeps token out of kubectl argv and sends it on stdin" {
  __vault_exec_kubectl() {
    [[ "${1:-}" == "--exec-stdin" ]] && shift
    printf '%s\n' "$*" >"$ARGV_LOG"
    cat >"$STDIN_LOG"
    return 0
  }
  _VAULT_SESSION_TOKENS["secrets/vault"]="hvs.session"

  run _vault_exec "secrets" "vault kv get secret/x" "vault"
  [ "${status}" -eq 0 ]
  run cat "$ARGV_LOG"
  [[ "${output}" != *"VAULT_TOKEN="* ]]
  [[ "${output}" == *"read -r __VAULT_TOKEN"* ]]
  run cat "$STDIN_LOG"
  [ "${output}" = "hvs.session" ]
}

@test "_vault_exec_stream --stdin multiplexes token line then payload" {
  __vault_exec_kubectl() {
    [[ "${1:-}" == "--exec-stdin" ]] && shift
    printf '%s\n' "$*" >"$ARGV_LOG"
    cat >"$STDIN_LOG"
    return 0
  }
  _VAULT_SESSION_TOKENS["secrets/vault"]="hvs.session"

  printf 'policy-body\n' | _vault_exec_stream --no-exit --stdin "secrets" "vault" \
    -- vault policy write eso-reader -
  run cat "$ARGV_LOG"
  [[ "${output}" != *"VAULT_TOKEN="* ]]
  [[ "${output}" == *"sh -c "* ]]
  [[ "${output}" == *"vault policy write eso-reader -"* ]]
  run cat "$STDIN_LOG"
  [ "${lines[0]}" = "hvs.session" ]
  [ "${lines[1]}" = "policy-body" ]
}

@test "_vault_exec_stream session branch without --stdin sends only token line" {
  __vault_exec_kubectl() {
    [[ "${1:-}" == "--exec-stdin" ]] && shift
    printf '%s\n' "$*" >"$ARGV_LOG"
    cat >"$STDIN_LOG"
    return 0
  }
  _VAULT_SESSION_TOKENS["secrets/vault"]="hvs.session"

  _vault_exec_stream --no-exit "secrets" "vault" -- vault write auth/kubernetes/role/r x=y
  run cat "$ARGV_LOG"
  [[ "${output}" != *"VAULT_TOKEN="* ]]
  [[ "${output}" == *"vault write auth/kubernetes/role/r x=y"* ]]
  run cat "$STDIN_LOG"
  [ "${output}" = "hvs.session" ]
}

@test "__vault_exec_kubectl --exec-stdin replays stdin across a retry" {
  ATTEMPT_FILE="$BATS_TEST_TMPDIR/attempts"
  printf '0' >"$ATTEMPT_FILE"
  _kubectl() {
    local seen; seen=$(cat)          # consume replayed stdin
    printf '%s' "$seen" >>"$STDIN_LOG"
    local n; n=$(cat "$ATTEMPT_FILE")
    if [[ "$n" == "0" ]]; then
      printf '1' >"$ATTEMPT_FILE"
      printf 'Error from server: container not found'
      return 1
    fi
    printf 'ok'
    return 0
  }

  run __vault_exec_kubectl --exec-stdin 0 "secrets" "vault" <<<"payload-line"
  [ "${status}" -eq 0 ]
  [ "${output}" = "ok" ]
  # payload was delivered on BOTH attempts (retry replay)
  run cat "$STDIN_LOG"
  [ "${output}" = "payload-linepayload-line" ]
}
```
